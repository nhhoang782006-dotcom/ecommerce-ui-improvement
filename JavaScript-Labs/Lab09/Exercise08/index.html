form.addEventListener("submit", (event) => {
  event.preventDefault();
  const missing = requiredFields.filter((field) => !field.value.trim());
  if (missing.length) {
    alert("Vui lòng điền các trường còn thiếu.");
    showErrors(missing);
  }
});
